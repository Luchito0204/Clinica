package com.utp.edu.pe.proyectoclinica.Singleton;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLDataException;
import java.sql.SQLException;

public class SingletonConexionDB {
    private static final String URL="jdbc:postgresql://localhost:5433/Clinica";
    private static final String USER="Lucho";
    private static final String PASSWORD="Clinica123";

    private static SingletonConexionDB instancia;

    private Connection connection;

    private SingletonConexionDB(){
        try{
            this.connection = DriverManager.getConnection(URL, USER, PASSWORD);
            System.out.println("SingletonDataBaseConnection: Conexion realizada con exito!");
        } catch (SQLDataException e){
            System.out.println("SingletonDataBaseConnection: Error al conectar a la base de datos!!");
        } catch (SQLException e) {
            throw new RuntimeException(e);
        }
    }

    public static SingletonConexionDB getInstancia(){
        if(instancia == null){
            instancia = new SingletonConexionDB();
        }
        return instancia;
    }

    public Connection getConnection(){
        return connection;
    }
}
